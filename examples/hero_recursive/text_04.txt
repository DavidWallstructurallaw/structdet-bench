# FIXTURE ONLY. Passive ordered-run merge illustration.
def sort_values(values):
    if len(values) < 2:
        return list(values)
    cut = len(values)//2
    left, right = sort_values(values[:cut]), sort_values(values[cut:])
    out = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            out.append(left[i]); i += 1
        else:
            out.append(right[j]); j += 1
    return out + left[i:] + right[j:]
