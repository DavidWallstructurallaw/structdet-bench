def sort_values(values):
    result = list(values)
    for end in range(len(result)-1, 0, -1):
        changed = False
        for j in range(end):
            if result[j] > result[j+1]:
                result[j], result[j+1] = result[j+1], result[j]
                changed = True
        if not changed:
            break
    return result
